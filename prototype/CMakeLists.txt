cmake_minimum_required(VERSION 3.22)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" ICEPUNK_VERSION LIMIT_COUNT 1)
project(ICEPUNKPrototype VERSION ${ICEPUNK_VERSION} LANGUAGES C CXX)
option(ICEPUNK_DEVELOPMENT_ENGINE "Allow source Python bridge in development builds" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(ICEPUNK_DEVELOPMENT_ENGINE)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
endif()
include(FetchContent)
FetchContent_Declare(JUCE
    GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG 91ad83ae34a81e0833b1a2b0866f54846370ae53)
FetchContent_MakeAvailable(JUCE)
juce_add_gui_app(ICEPUNK PRODUCT_NAME "ICEPUNK" COMPANY_NAME "ICEPUNK"
    BUNDLE_ID "audio.icepunk.midigenerator" VERSION ${PROJECT_VERSION}
    ICON_BIG "${CMAKE_CURRENT_SOURCE_DIR}/../assets/icepunk-snowflake.png"
    ICON_SMALL "${CMAKE_CURRENT_SOURCE_DIR}/../assets/icepunk-snowflake.png")
juce_generate_juce_header(ICEPUNK)
if(ICEPUNK_DEVELOPMENT_ENGINE)
    set_target_properties(ICEPUNK PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../app")
else()
    set_target_properties(ICEPUNK PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}/release-app")
endif()
if(MSVC)
    set_property(TARGET ICEPUNK PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
target_sources(ICEPUNK PRIVATE src/Main.cpp src/GeneratorEngine.cpp src/MidiPreview.cpp src/MainComponent.cpp src/RuntimePaths.cpp src/LicenseManager.cpp vendor/monocypher/monocypher.c vendor/monocypher/monocypher-ed25519.c)
target_compile_definitions(ICEPUNK PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
    JUCE_APPLICATION_NAME_STRING="$<TARGET_PROPERTY:ICEPUNK,JUCE_PRODUCT_NAME>"
    JUCE_APPLICATION_VERSION_STRING="${PROJECT_VERSION}"
    ICEPUNK_DEVELOPMENT_ENGINE=$<BOOL:${ICEPUNK_DEVELOPMENT_ENGINE}>)
target_link_libraries(ICEPUNK PRIVATE juce::juce_gui_basics juce::juce_audio_basics juce::juce_cryptography
    juce::juce_recommended_config_flags)
set(ICEPUNK_LICENSE_ORIGIN "" CACHE STRING "HTTPS licensing server origin")
set(ICEPUNK_LICENSE_PUBLIC_KEY "" CACHE STRING "Ed25519 raw public key, 64 lowercase hex characters")
if(NOT ICEPUNK_LICENSE_ORIGIN MATCHES "^https://[a-zA-Z0-9.-]+(:[0-9]+)?$" AND NOT ICEPUNK_LICENSE_ORIGIN MATCHES "^http://(localhost|127[.]0[.]0[.]1):[0-9]+$")
    message(FATAL_ERROR "Set ICEPUNK_LICENSE_ORIGIN to an HTTPS origin (HTTP loopback is Debug only)")
endif()
string(LENGTH "${ICEPUNK_LICENSE_PUBLIC_KEY}" key_length)
if(NOT key_length EQUAL 64 OR NOT ICEPUNK_LICENSE_PUBLIC_KEY MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "Set ICEPUNK_LICENSE_PUBLIC_KEY to 64 lowercase hex characters")
endif()
configure_file(AppConfig.h.in generated/AppConfig.h @ONLY)
target_include_directories(ICEPUNK PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/monocypher")
enable_testing()
if(ICEPUNK_DEVELOPMENT_ENGINE)
    add_test(NAME PythonIntegration COMMAND "${Python3_EXECUTABLE}" -m unittest discover
        -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p "test_*.py" -v)
endif()

option(ICEPUNK_BUILD_LICENSE_TESTS "Build separate licensing test executable (never packaged)" OFF)
if(ICEPUNK_BUILD_LICENSE_TESTS)
    juce_add_console_app(LicenseTests PRODUCT_NAME "LicenseTests")
    juce_generate_juce_header(LicenseTests)
    target_sources(LicenseTests PRIVATE tests/license_tests.cpp src/LicenseManager.cpp src/RuntimePaths.cpp vendor/monocypher/monocypher.c vendor/monocypher/monocypher-ed25519.c)
    target_include_directories(LicenseTests PRIVATE src vendor/monocypher "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(LicenseTests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 ICEPUNK_DEVELOPMENT_ENGINE=0)
    target_link_libraries(LicenseTests PRIVATE juce::juce_core juce::juce_cryptography)
    if(MSVC)
        set_property(TARGET LicenseTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endif()
